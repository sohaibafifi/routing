cmake_minimum_required(VERSION 3.15)
project(routing LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Set macOS architectures automatically when building on Apple platforms.
if(APPLE)
  if(NOT DEFINED CMAKE_OSX_ARCHITECTURES OR CMAKE_OSX_ARCHITECTURES STREQUAL "")
    if(DEFINED CMAKE_HOST_SYSTEM_PROCESSOR)
      string(TOLOWER "${CMAKE_HOST_SYSTEM_PROCESSOR}" _proc)
      if(_proc MATCHES "^(aarch64|arm64|armv8)")
        set(_osx_arch "arm64")
      elseif(_proc MATCHES "^(x86_64|amd64)")
        set(_osx_arch "x86_64")
      else()
        set(_osx_arch "x86_64;arm64")
      endif()
    else()
      set(_osx_arch "x86_64;arm64")
    endif()
    set(CMAKE_OSX_ARCHITECTURES "${_osx_arch}" CACHE STRING "Auto-detected macOS architectures" FORCE)
  endif()
endif()

option(ROUTING_VERBOSE_MAKEFILE "Enable verbose build output" OFF)
if(ROUTING_VERBOSE_MAKEFILE)
  set(CMAKE_VERBOSE_MAKEFILE on)
endif()

option(ROUTING_BUILD_EXAMPLES "Build examples" ON)
option(ROUTING_BUILD_CPLEX "Build CPLEX backends (MIP and CP Optimizer)" ON)
option(ROUTING_BUILD_CPOPTIMIZER "Build the cpoptimizer CLI (requires CPLEX)" ON)
option(ROUTING_BUILD_GRB "Build the grb CLI (requires GUROBI)" ON)
option(ROUTING_BUILD_XCSP3 "Build XCSP3 backend (requires ACE/CoSoCo at runtime)" ON)
option(ROUTING_BUILD_ORTOOLS "Build OR-Tools CP-SAT backend" ON)
option(ROUTING_BUILD_PYTHON "Build Python bindings (requires Python and nanobind)" OFF)

include(CTest)

list(PREPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")

# Helpers
function(routing_target_link_dl target_name)
  if(CMAKE_DL_LIBS)
    target_link_libraries(${target_name} PUBLIC ${CMAKE_DL_LIBS})
  endif()
endfunction()

# Find CPLEX (optional - disable with -DROUTING_BUILD_CPLEX=OFF)
if(ROUTING_BUILD_CPLEX)
  find_package(Cplex QUIET)
  if(CPLEX_FOUND)
    message(STATUS "Found CPLEX: ${CPLEX_LIBRARIES}")
    add_compile_definitions(CPLEX_FOUND)
    include_directories(SYSTEM ${CPLEX_INCLUDE_DIRS})
  else()
    message(STATUS "CPLEX not found - CPLEX backends will not be available")
  endif()
else()
  message(STATUS "CPLEX disabled by user (-DROUTING_BUILD_CPLEX=OFF)")
endif()

find_package(GUROBI)
if(GUROBI_FOUND)
  message(STATUS "GUROBI_LIBRARY: ${GUROBI_LIBRARY}")
  message(STATUS "GUROBI_CXX_LIBRARY: ${GUROBI_CXX_LIBRARY}")
  add_compile_definitions(GUROBI_FOUND)
  include_directories(SYSTEM ${GUROBI_INCLUDE_DIRS})
else()
  message(STATUS "Gurobi not found - Gurobi backend will not be available")
endif()

# Find HiGHS (optional open-source MIP solver)
find_package(HiGHS QUIET)
if(HIGHS_FOUND)
  message(STATUS "Found HiGHS: ${HIGHS_LIBRARIES}")
  add_compile_definitions(HIGHS_FOUND)
  include_directories(SYSTEM ${HIGHS_INCLUDE_DIRS})
else()
  message(STATUS "HiGHS not found - HiGHS backend will not be available")
endif()

# Find OR-Tools (optional open-source optimization library)
if(ROUTING_BUILD_ORTOOLS)
  find_package(ORTools QUIET)
  if(ORTOOLS_FOUND)
    message(STATUS "Found OR-Tools: ${ORTOOLS_LIBRARIES}")
    add_compile_definitions(ORTOOLS_FOUND)
    include_directories(SYSTEM ${ORTOOLS_INCLUDE_DIRS})
  else()
    message(STATUS "OR-Tools not found - CP-SAT backend will not be available")
  endif()
else()
  message(STATUS "OR-Tools disabled by user (-DROUTING_BUILD_ORTOOLS=OFF)")
endif()

add_subdirectory(routing)
add_subdirectory(plugins)

if(ROUTING_BUILD_EXAMPLES)
  add_subdirectory(examples)
endif()

if(BUILD_TESTING)
  add_subdirectory(tests)
endif()

if(ROUTING_BUILD_PYTHON)
  add_subdirectory(python)
endif()
